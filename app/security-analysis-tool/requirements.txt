# get_secret/create_secret/update_secret taking full_name) first ships in
# 0.105.0. 0.104 and earlier lack the class entirely, so secrets_uc.* would raise
# AttributeError at runtime. The alerts/v2 API (WorkspaceClient.alerts_v2, the
# sql.AlertV2* models) needed for secret-scanning alerts arrived earlier at 0.51
# and is comfortably covered by this higher floor, as are jobs, serving
# endpoints, statement execution, notification destinations and genie.

werkzeug~=3.0
databricks-sdk>=0.105,<1.0
